# LumenBox M0 技术验证 · 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.（当前环境未安装上述技能，采用会话内逐任务执行 + 每任务验证即可。）

**Goal:** 用一个可运行的 spike 程序（`lumenbox.exe`）实测验证 C++ / Win32 + Direct2D + Windows.UI.Composition 路线的四项关键风险——透明玻璃、WorkerW 贴桌、合成线程弹簧动效、极低资源占用——并以实测数据锁定技术路线（设计文档 D1 门禁）。

**Tech Stack:** C++23（VS2022 MSVC）、Win32/DWM、Direct2D + DirectWrite、C++/WinRT（Windows.UI.Composition）、nlohmann/json（配置）、Catch2（测试）、CMake ≥ 3.25。

**Spec:** `docs/设计文档.md` v0.3（本计划覆盖其 M0 行 + §6.1/§6.2/§7.2 关键技术验证 + §8 测量协议；M1 计划在 M0 门禁通过后另行编写）。

- [ ] **Step 4: 实现 config.cpp**

```cpp
#include "core/config.hpp"
#include <nlohmann/json.hpp>
#include <shlobj.h>

namespace lb {
namespace fs = std::filesystem;
namespace json = nlohmann;

SpikeConfig ConfigStore::load() const {
    SpikeConfig c;
    std::error_code ec;
    if (!fs::exists(file_, ec)) return c;
    try {
        json::json j = json::json::parse(std::ifstream(file_, std::ios::binary));
        c.mode = j.value("mode", c.mode);
        c.x = j.value("x", c.x);
        c.y = j.value("y", c.y);
        c.w = j.value("w", c.w);
        c.h = j.value("h", c.h);
        if (c.mode != "independent" && c.mode != "desktop" && c.mode != "bottom")
            c.mode = "independent";
    } catch (...) {
        return SpikeConfig{};  // 损坏即回默认，绝不阻塞启动
    }
    return c;
}

bool ConfigStore::save(const SpikeConfig& c) const {
    json::json j{{"mode", c.mode}, {"x", c.x}, {"y", c.y}, {"w", c.w}, {"h", c.h}};
    std::error_code ec;
    fs::create_directories(file_.parent_path(), ec);
    auto tmp = file_.wstring() + L".tmp";
    { std::ofstream f(tmp, std::ios::binary); f << j.dump(2); }
    fs::rename(tmp, file_, ec);  // std::filesystem::rename 在 Windows 上原子替换已有目标
    return !ec;
}

fs::path ConfigStore::default_dir() {
    PWSTR raw = nullptr;
    if (FAILED(SHGetKnownFolderPath(FOLDERID_RoamingAppData, 0, nullptr, &raw)))
        return L".";
    fs::path dir(raw);
    CoTaskMemFree(raw);
    return dir / L"LumenBox";
}

}  // namespace lb
```

```cpp
    void OnSize(int w, int h);
```

- [ ] **Step 2: 写 glass_window.cpp**

```cpp
#include "ui/glass_window.hpp"
#include "core/config.hpp"
#include <dwmapi.h>
#pragma comment(lib, "dwmapi")

#ifndef DWMWA_SYSTEMBACKDROP_TYPE
#define DWMWA_SYSTEMBACKDROP_TYPE 38
#endif
#ifndef DWMSBT_TRANSIENTWINDOW
#define DWMSBT_TRANSIENTWINDOW 3  // Acrylic
#endif

namespace lb {

static bool IsWin11OrGreater() {
    // RtlGetVersion 不受兼容性清单影响，返回真实版本
    OSVERSIONINFOW vi{};
    vi.dwOSVersionInfoSize = sizeof(vi);
    using RtlGetVersionFn = LONG(WINAPI*)(OSVERSIONINFOW*);
    auto fn = reinterpret_cast<RtlGetVersionFn>(
        GetProcAddress(GetModuleHandleW(L"ntdll.dll"), "RtlGetVersion"));
    return fn && fn(&vi) == 0 && vi.dwBuildNumber >= 22000;
}

bool GlassWindow::Create(HINSTANCE hInst, const std::wstring& title,
                         SpikeConfig& cfg, std::function<void(UINT)> on_key) {
    cfg_ = &cfg;
    on_key_ = std::move(on_key);

    WNDCLASSEXW wc{};
    wc.cbSize = sizeof(wc);
    wc.hCursor = LoadCursorW(nullptr, IDC_ARROW);
    wc.lpfnWndProc = &GlassWindow::WndProc;
    wc.hInstance = hInst;
    wc.lpszClassName = L"LumenBoxGlass";
    RegisterClassExW(&wc);

    hwnd_ = CreateWindowExW(WS_EX_NOREDIRECTIONBITMAP | WS_EX_TOOLWINDOW,
                            wc.lpszClassName, title.c_str(), WS_POPUP,
                            cfg.x, cfg.y, cfg.w, cfg.h, nullptr, nullptr, hInst, this);
    if (!hwnd_) return false;
    dpi_ = static_cast<float>(GetDpiForWindow(hwnd_));

    compositor_ = winrt::Windows::UI::Composition::Compositor();
    target_ = compositor_.CreateDesktopWindowTarget(hwnd_, /*isTopmost=*/false);

    MARGINS m{-1};  // 让 DWM 玻璃铺满整个客户区
    DwmExtendFrameIntoClientArea(hwnd_, &m);
    ApplyBackdrop();

    OnSize(cfg.w, cfg.h);
    ShowWindow(hwnd_, SW_SHOW);
    return true;
}

void GlassWindow::ApplyBackdrop() {
    if (IsWin11OrGreater()) {
        INT32 backdrop = DWMSBT_TRANSIENTWINDOW;
        if (SUCCEEDED(DwmSetWindowAttribute(hwnd_, DWMWA_SYSTEMBACKDROP_TYPE,
                                            &backdrop, sizeof(backdrop))))
            return;
    }
    // Win10 退化：SetWindowCompositionAttribute（未公开 API，动态加载）
    struct ACCENT_POLICY { INT32 State; INT32 Flags; UINT32 GradientColor; INT32 AnimationId; };
    struct WCA_DATA { INT32 Attrib; PVOID pvData; SIZE_T cbData; };
    using SetWCAFN = BOOL(WINAPI*)(HWND, const WCA_DATA*);
    auto fn = reinterpret_cast<SetWCAFN>(
        GetProcAddress(GetModuleHandleW(L"user32.dll"), "SetWindowCompositionAttribute"));
    if (fn) {
        constexpr INT32 ACCENT_ENABLE_ACRYLICBLURBEHIND = 4;
        ACCENT_POLICY p{ACCENT_ENABLE_ACRYLICBLURBEHIND, 2, 0xB0101010, 0};  // ABGR
        WCA_DATA d{19 /*WCA_ACCENT_POLICY*/, &p, sizeof(p)};
        fn(hwnd_, &d);
    }
    // 三级退化：不做任何系统模糊，卡片自绘底色保证可读
}

LRESULT GlassWindow::WndProc(HWND hwnd, UINT msg, WPARAM wp, LPARAM lp) {
    auto* self = reinterpret_cast<GlassWindow*>(GetWindowLongPtrW(hwnd, GWLP_USERDATA));
    switch (msg) {
    case WM_NCCREATE:
        SetWindowLongPtrW(hwnd, GWLP_USERDATA,
            reinterpret_cast<LONG_PTR>(
                reinterpret_cast<CREATESTRUCTW*>(lp)->lpCreateParams));
        return DefWindowProcW(hwnd, msg, wp, lp);
    case WM_NCHITTEST: return HTCAPTION;  // spike：整窗可拖
    case WM_KEYDOWN:
        if (wp == VK_ESCAPE) DestroyWindow(hwnd);
        return 0;
    case WM_HOTKEY:
        if (self && self->on_key_) self->on_key_(static_cast<UINT>(wp));
        return 0;
    case WM_SIZE:
        if (self) self->OnSize(LOWORD(lp), HIWORD(lp));
        return 0;
    case WM_DESTROY:
        if (self) self->SaveBounds();
        PostQuitMessage(0);
        return 0;
    }
    return DefWindowProcW(hwnd, msg, wp, lp);
}

void GlassWindow::OnSize(int w, int h) {
    if (!root_) root_ = compositor_.CreateSpriteVisual();  // 全透明根：玻璃由此透出
    root_.Size({static_cast<float>(w), static_cast<float>(h)});
    target_.Root(root_);
}

void GlassWindow::SaveBounds() {
    RECT r{};
    GetWindowRect(hwnd_, &r);
    cfg_->x = r.left;
    cfg_->y = r.top;
    cfg_->w = r.right - r.left;
    cfg_->h = r.bottom - r.top;
}

}  // namespace lb
```

- [ ] **Step 2: 写 card_visual.cpp（Aurora Glass 首个真实实现：白 7% 卡面 + 白 12% 内发光 + 极光渐变描边）**

```cpp
#include "ui/card_visual.hpp"
#include <d2d1_2.h>
#include <dwrite.h>
#include <windows.ui.composition.interop.h>
#pragma comment(lib, "d2d1")
#pragma comment(lib, "dwrite")

namespace lb {
namespace wuc = winrt::Windows::UI::Composition;

SpriteVisual CreateCardVisual(Compositor const& compositor,
                              float wDip, float hDip, float dpiScale) {
    UINT32 pxW = static_cast<UINT32>(wDip * dpiScale);
    UINT32 pxH = static_cast<UINT32>(hDip * dpiScale);

    auto interop = compositor.as<::ICompositorInterop>();
    winrt::com_ptr<::ICompositionDrawingSurface> surface;
    winrt::check_hresult(interop->CreateDrawingSurface(
        {static_cast<FLOAT>(pxW), static_cast<FLOAT>(pxH)},
        DXGI_FORMAT_B8G8R8A8_UNORM, DXGI_ALPHA_MODE_PREMULTIPLIED, surface.put()));

    winrt::com_ptr<::ID2D1DeviceContext> ctx;
    POINT offset{};
    RECT update{0, 0, static_cast<LONG>(pxW), static_cast<LONG>(pxH)};
    winrt::check_hresult(
        surface.as<::ICompositionDrawingSurfaceInterop>()->BeginDraw(
            &update, winrt::guid_of<::ID2D1DeviceContext>(), ctx.put_void(), &offset));

    ctx->SetTransform(
        D2D1::Matrix3x2F::Translation(static_cast<float>(offset.x),
                                      static_cast<float>(offset.y)) *
        D2D1::Matrix3x2F::Scale(D2D1::SizeF(dpiScale, dpiScale), D2D1::Point2F(0, 0)));
    ctx->Clear(nullptr);  // 全透明底：DWM 玻璃从此透出

    const float inset = 8.f, r = 16.f;
    const D2D1_RECT_F card{inset, inset, wDip - inset, hDip - inset};
```

```cpp
#include "shell/desktop_layer.hpp"

namespace lb {

static HWND FindWorkerWForDesktop() {
    HWND progman = FindWindowW(L"Progman", nullptr);
    if (!progman) return nullptr;
    // 0x052C：让系统在壁纸与图标之间生成/刷新 WorkerW（Wallpaper Engine 同款手法）
    SendMessageTimeoutW(progman, 0x052C, 0, 0, SMTO_NORMAL, 1000, nullptr);

    struct Ctx {
        HWND workerAfter = nullptr;
    };
    Ctx ctx;
    EnumWindows(
        +[](HWND hwnd, LPARAM lp) -> BOOL {
            if (FindWindowExW(hwnd, nullptr, L"SHELLDLL_DefView", nullptr)) {
                // 图标层 WorkerW 之后的下一个 WorkerW 才是壁纸层
                auto* c = reinterpret_cast<Ctx*>(lp);
                c->workerAfter = FindWindowExW(nullptr, hwnd, L"WorkerW", nullptr);
                return FALSE;
            }
            return TRUE;
        },
        reinterpret_cast<LPARAM>(&ctx));
    return ctx.workerAfter;
}

bool AttachToDesktopLayer(HWND hwnd) {
    HWND worker = FindWorkerWForDesktop();
    if (!worker) return false;
    LONG_PTR style = GetWindowLongPtrW(hwnd, GWL_STYLE);
    SetWindowLongPtrW(hwnd, GWL_STYLE, (style & ~WS_POPUP) | WS_CHILD);
    if (!SetParent(hwnd, worker)) {
        SetWindowLongPtrW(hwnd, GWL_STYLE, style);  // 失败还原样式
        return false;
    }
    SetWindowPos(hwnd, nullptr, 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE | SWP_FRAMECHANGED);
    return true;
}

void UseBottomMode(HWND hwnd) {
    LONG_PTR style = GetWindowLongPtrW(hwnd, GWL_STYLE);
    SetWindowLongPtrW(hwnd, GWL_STYLE, (style & ~WS_CHILD) | WS_POPUP);
    SetParent(hwnd, nullptr);
    SetWindowPos(hwnd, HWND_BOTTOM, 0, 0, 0, 0,
                 SWP_NOMOVE | SWP_NOSIZE | SWP_FRAMECHANGED);
}

bool RestoreIndependent(HWND hwnd) {
    LONG_PTR style = GetWindowLongPtrW(hwnd, GWL_STYLE);
    SetWindowLongPtrW(hwnd, GWL_STYLE, (style & ~WS_CHILD) | WS_POPUP);
    if (!SetParent(hwnd, nullptr)) return false;
    SetWindowPos(hwnd, HWND_TOP, 0, 0, 0, 0,
                 SWP_NOMOVE | SWP_NOSIZE | SWP_FRAMECHANGED);
    return true;
}

}  // namespace lb
```

```cpp
namespace lb {
namespace wuc = winrt::Windows::UI::Composition;

static wuc::CompositionSurfaceBrush MakeDotBrush(Compositor const& compositor) {
    auto interop = compositor.as<::ICompositorInterop>();
    winrt::com_ptr<::ICompositionDrawingSurface> surface;
    winrt::check_hresult(interop->CreateDrawingSurface(
        {64.f, 64.f}, DXGI_FORMAT_B8G8R8A8_UNORM, DXGI_ALPHA_MODE_PREMULTIPLIED,
        surface.put()));

    winrt::com_ptr<::ID2D1DeviceContext> ctx;
    POINT offset{};
    RECT all{0, 0, 64, 64};
    winrt::check_hresult(
        surface.as<::ICompositionDrawingSurfaceInterop>()->BeginDraw(
            &all, winrt::guid_of<::ID2D1DeviceContext>(), ctx.put_void(), &offset));
    ctx->SetTransform(D2D1::Matrix3x2F::Translation(static_cast<float>(offset.x),
                                                    static_cast<float>(offset.y)));

    D2D1_GRADIENT_STOP gs[2] = {
        {0.0f, D2D1_COLOR_F{0.506f, 0.549f, 0.973f, 1.0f}},  // 靛 #818CF8
        {1.0f, D2D1_COLOR_F{0.506f, 0.549f, 0.973f, 0.0f}},
    };
    winrt::com_ptr<::ID2D1GradientStopCollection> stops;
    ctx->CreateGradientStopCollection(gs, 2, D2D1_GAMMA_2_2,
                                      D2D1_EXTEND_MODE_CLAMP, stops.put());
    winrt::com_ptr<::ID2D1RadialGradientBrush> grad;
    ctx->CreateRadialGradientBrush(
        D2D1::RadialGradientBrushProperties(D2D1::Point2F(32, 32), D2D1::Point2F(0, 0),
                                            D2D1::Point2F(32, 32)),
        stops.get(), grad.put());
    ctx->FillEllipse(D2D1_ELLIPSE{{32, 32}, 32, 32}, grad.get());
    winrt::check_hresult(
        surface.as<::ICompositionDrawingSurfaceInterop>()->EndDraw());

    return compositor.CreateSurfaceBrush(surface.as<wuc::ICompositionSurface>());
}

void SpringDemo::Init(Compositor const& compositor, SpriteVisual const& root,
                      float wDip, float hDip) {
    w_ = wDip;
    h_ = hDip;
    auto brush = MakeDotBrush(compositor);
    for (int i = 0; i < 3; ++i) {
        auto dot = compositor.CreateSpriteVisual();
        dot.Brush(brush);
        dot.Size({32, 32});
        dot.Offset({60.f + i * 90.f, h_ * 0.55f, 0.f});
        root.Children().InsertAtTop(dot);
        dots_.push_back(dot);
    }
}

void SpringDemo::Trigger() {
    if (dots_.empty()) return;
    auto compositor = dots_[0].Compositor();
    flipped_ = !flipped_;
    for (size_t i = 0; i < dots_.size(); ++i) {
        // 每个点独立动画实例（DelayTime 各异，参数不能共享）
        auto spring = compositor.CreateSpringVector2Animation();
        spring.DampingRatio(0.9f);  // ≤3% 微过冲（设计 §7.2）
        spring.Period(std::chrono::milliseconds(40));
        spring.InitialVelocity({0.f, 0.f});
        float x = flipped_ ? w_ - 92.f - static_cast<float>(i) * 90.f
                           : 60.f + static_cast<float>(i) * 90.f;
        spring.FinalValue({x, h_ * 0.55f});
        spring.DelayTime(std::chrono::milliseconds(20 * static_cast<int>(i)));
        spring.DelayBehavior(wuc::AnimationDelayBehavior::SetInitialValueBeforeDelay);
        dots_[i].StartAnimation(L"Offset", spring);
    }
}

}  // namespace lb
```

```powershell
# scripts/measure.ps1 —— 采样 lumenbox 进程资源占用
# 用法: powershell -ExecutionPolicy Bypass -File scripts\measure.ps1 [-Seconds 60]
param([string]$Name = "lumenbox", [int]$Seconds = 60)

$samples = @()
$end = (Get-Date).AddSeconds($Seconds)
while ((Get-Date) -lt $end) {
    $p = Get-Process -Name $Name -ErrorAction SilentlyContinue
    if ($p) {
        $samples += [pscustomobject]@{
            Time         = Get-Date -Format HH:mm:ss
            PrivateMB    = [math]::Round($p.PrivateMemorySize64 / 1MB, 1)
            WorkingSetMB = [math]::Round($p.WorkingSet64 / 1MB, 1)
            Handles      = $p.HandleCount
            CPUsec       = [math]::Round($p.TotalProcessorTime.TotalSeconds, 2)
        }
    }
    Start-Sleep -Seconds 5
}
$samples | Format-Table -AutoSize
$samples | Export-Csv -Path "$PSScriptRoot\..\docs\m0-measure.csv" -NoTypeInformation -Encoding UTF8
Write-Host "空闲 CPU ≈ $([math]::Round(($samples[-1].CPUsec - $samples[0].CPUsec) / $Seconds * 100, 2)) %"
```
